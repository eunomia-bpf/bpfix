#ifndef __TARGET_ARCH_x86
#define __TARGET_ARCH_x86 1
#endif

#include <vmlinux.h>
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_endian.h>

#ifndef XDP_PASS
#define XDP_PASS 2
#endif
#ifndef XDP_DROP
#define XDP_DROP 1
#endif
#ifndef ETH_P_IP
#define ETH_P_IP 0x0800
#endif
#ifndef IPPROTO_UDP
#define IPPROTO_UDP 17
#endif

SEC("xdp")
int alu32_pointer_cookie(struct xdp_md *ctx)
{
    void *data = (void *)(long)ctx->data;
    void *data_end = (void *)(long)ctx->data_end;
    struct ethhdr *eth = data;
    __u64 cookie;

    if ((void *)(eth + 1) > data_end)
        return XDP_PASS;
    if (bpf_ntohs(eth->h_proto) != ETH_P_IP)
        return XDP_PASS;

    struct iphdr *iph = data + sizeof(*eth);
    if ((void *)(iph + 1) > data_end)
        return XDP_PASS;
    if (iph->protocol != IPPROTO_UDP)
        return XDP_PASS;

    cookie = (__u64)(long)iph;
    // Perform the shift on the scalar cookie, not on the pointer register directly.
    // The verifier rejects pointer arithmetic with <<.
    // We cast to __u64, shift, then cast back to pointer for addition.
    asm volatile("%[cookie] <<= 32; %[cookie] >>= 32" : [cookie] "+r"(cookie));

    // Use cookie (scalar) for pointer arithmetic instead of iph (pointer)
    struct udphdr *udp = (void *)(long)(cookie + sizeof(*iph));
    if ((void *)(udp + 1) > data_end)
        return XDP_PASS;

    return bpf_ntohs(udp->dest) == 53 ? XDP_DROP : XDP_PASS;
}
